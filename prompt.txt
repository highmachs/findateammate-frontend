tsimport { describe, it, expect } from "vitest";

const BASE = "http://localhost:3000";

async function get(path: string, cookie?: string) {
  return fetch(`${BASE}${path}`, {
    headers: cookie ? { cookie } : {},
    credentials: "include",
  });
}

async function post(path: string, body: object, cookie?: string) {
  return fetch(`${BASE}${path}`, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      ...(cookie ? { cookie } : {}),
    },
    body: JSON.stringify(body),
    credentials: "include",
  });
}

    const roomId = "test-chat-" + Date.now();
    const sender = new PartySocket({ host: "localhost:1999", room: roomId, party: "chat" });
    const receiver = new PartySocket({ host: "localhost:1999", room: roomId, party: "chat" });

    await new Promise<void>((r) => receiver.addEventListener("open", () => r()));
    await new Promise<void>((r) => sender.addEventListener("open", () => r()));

    const received = new Promise<string>((resolve) => {
